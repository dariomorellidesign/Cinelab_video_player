cmake_minimum_required(VERSION 3.24)
project(DMPConfidenceTests LANGUAGES CXX)
enable_testing()
add_executable(ConfidenceRepairTests ConfidenceRepairTests.cpp)
target_compile_features(ConfidenceRepairTests PRIVATE cxx_std_20)
target_compile_definitions(ConfidenceRepairTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ConfidenceRepairTests PRIVATE /W4 /EHsc /O2)
target_include_directories(ConfidenceRepairTests PRIVATE ../src)
add_test(NAME ConfidenceRepairEquivalence COMMAND ConfidenceRepairTests)
add_executable(StreamlineFailureTests StreamlineFailureTests.cpp)
target_compile_features(StreamlineFailureTests PRIVATE cxx_std_20)
target_compile_definitions(StreamlineFailureTests PRIVATE DMP_TESTING NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(StreamlineFailureTests PRIVATE /W4 /EHsc)
target_include_directories(StreamlineFailureTests PRIVATE ../src ../external/streamline-sdk-v2.12.0/include)
add_test(NAME StreamlineFailureFallback COMMAND StreamlineFailureTests)
